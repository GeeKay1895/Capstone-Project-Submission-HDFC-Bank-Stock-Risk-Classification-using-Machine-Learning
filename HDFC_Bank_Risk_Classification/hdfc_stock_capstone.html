# Creating Risk Classes
conditions = [
    (df['Z_Score'] < -1),
    (df['Z_Score'] >= -1) & (df['Z_Score'] <= 1),
    (df['Z_Score'] > 1)
]

risk_labels = [
    'Low Risk',
    'Medium Risk',
    'High Risk'
]

df['Risk_Class'] = np.select(
    conditions,
    risk_labels,
    default='Medium Risk'
)
